// Пример функции для отправки уведомления в Telegram чат async function sendTelegramNotification(task) { const BOT_TOKEN = 'ВАШ_TELEGRAM_BOT_TOKEN'; const CHAT_ID = '@id_вашего_чата_или_канала'; // например, -100123456789 const message = ` ⚡ **Новый заказ в Profitum Chisinau!** 📌 **${task.title}** 📍 Сектор: ${task.district || 'Centru'} 💰 Бюджет: **${task.price} MDL** 📂 Категория: ${task.category || 'Услуги'} 👉 *Зайдите в мини-приложение, чтобы откликнуться!* `.trim(); const url = `https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`; await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: CHAT_ID, text: message, parse_mode: 'Markdown', disable_notification: false // false означает, что у пользователей прозвучит звуковой сигнал! }) }); }
