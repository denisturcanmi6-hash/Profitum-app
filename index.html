const { Telegraf, Markup } = require('telegraf'); const bot = new Telegraf('YOUR_BOT_TOKEN'); // База данных (в реальном проекте здесь будет MongoDB/PostgreSQL) // Храним ID создателя задания: { taskId: { title: '...', ownerId: 12345678 } } const tasksDB = { 'task_101': { title: 'Помощь с переездом в Кишиневе', ownerId: 98765432 } }; // 1. Исполнитель нажимает кнопку отклика на задание bot.action(/^apply_(.+)$/, async (ctx) => { const taskId = ctx.match[1]; const task = tasksDB[taskId]; const applicant = ctx.from; // Кто откликнулся (исполнитель) if (!task) { return ctx.reply('Задание не найдено или уже закрыто.'); } // Текст сообщения для заказчика const messageForOwner = `🔔 Новый отклик на ваше задание: "${task.title}"\n\n` + `👤 Исполнитель: ${applicant.first_name} (@${applicant.username || 'нет юзернейма'})\n` + `ID: ${applicant.id}`; try { // Отправляем уведомление заказчику с кнопкой связи await bot.telegram.sendMessage(task.ownerId, messageForOwner, { reply_markup: Markup.inlineKeyboard([ [Markup.button.url('💬 Написать исполнителю', `https://t.me/${applicant.username}`)], [Markup.button.callback('✅ Подтвердить сделку', `accept_${taskId}_${applicant.id}`)] ]).reply_markup }); await ctx.answerCbQuery('Отклик успешно отправлен заказчику!'); await ctx.reply('Ваш отклик отправлен заказчику. Ожидайте связи!'); } catch (error) { console.error(error); await ctx.reply('Не удалось связаться с заказчиком. Попробуйте позже.'); } }); bot.launch();
