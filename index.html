function changeLanguage(lang) { currentLang = lang; const t = translations[lang] || translations.ru; // Безопасное обновление текста с проверкой наличия атрибута document.querySelectorAll('[data-i18n]').forEach(el => { const key = el.getAttribute('data-i18n'); if (t[key]) { el.textContent = t[key]; } }); // Обновление плейсхолдеров document.querySelectorAll('[data-i18n-placeholder]').forEach(el => { const key = el.getAttribute('data-i18n-placeholder'); if (t[key]) { el.placeholder = t[key]; } }); // Обновление категорий document.querySelectorAll('[data-i18n-cat]').forEach(el => { const key = el.getAttribute('data-i18n-cat'); if (t[key]) { el.textContent = t[key]; } }); // Обновление секторов Кишинева в фильтре const filterDistrict = document.getElementById('filter-district'); if (filterDistrict && t.districts) { const currentVal = filterDistrict.value; filterDistrict.innerHTML = ` `; filterDistrict.value = currentVal; } // Синхронизируем состояние селекта языков во всех возможных селекторах document.querySelectorAll('#lang-switcher').forEach(sel => { sel.value = lang; }); renderTasks(tasksList); renderUserProfile(); if (!document.getElementById('section-executors').classList.contains('hidden')) { renderExecutors(); } }
